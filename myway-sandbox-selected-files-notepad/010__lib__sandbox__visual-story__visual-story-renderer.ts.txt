// -----------------------------------------------------------------------------
// MyWay Visual Story Renderer Schema
// -----------------------------------------------------------------------------
//
// Purpose:
// A Visual Story is a scene-first teaching contract. The orientation script is
// the source of truth. The visual plan, timeline, camera motion, and post-story
// interaction are derived from that same script instead of from a separate
// staging paragraph or recap.
//
// This schema is intentionally renderer-facing, not provider-specific. Models
// draft it; MyWay validates, normalizes, and renders it using trusted scene
// primitives and interaction controls.

export type VisualStoryRendererSchemaVersion = "myway_visual_story_renderer_v1";

export type VisualStorySceneKind = "visual_story";

export type VisualStoryRendererParams = {
  schema_version: VisualStoryRendererSchemaVersion;
  scene_kind: VisualStorySceneKind;

  /**
   * The actual teaching experience. The story.orientation_script is the
   * canonical learner-facing script and must drive the visual plan.
   */
  story: VisualStoryStory;

  /**
   * What becomes explorable after the autoplay story finishes.
   */
  interaction_phase?: VisualStoryInteractionPhase | null;

  /**
   * Optional lightweight check after the visual story.
   * The story teaches first; the check measures after.
   */
  optional_check?: VisualStoryOptionalCheck | null;

  /**
   * Temporary compatibility mirror for the current VisualStoryProbe renderer.
   * New code should prefer story.orientation_script and story.beats, but the
   * current renderer can still read these top-level fields while migrating.
   */
  canonical_orientation_script?: string;
  title?: string;
  learning_goal?: string;
  scene_family?: VisualStorySceneFamily;
  script_beats?: VisualStoryCompatScriptBeat[];
  scene_objects?: VisualStoryCompatSceneObject[];
  scene_actions?: VisualStoryCompatSceneAction[];
  camera_plan?: VisualStoryCompatCameraBeat[];
  success_criteria?: VisualStorySuccessCriterion[];
};

export type VisualStoryStory = {
  /**
   * The authoritative learner-facing orientation paragraph.
   * Do not include recap bullets or a second staging explanation here.
   */
  orientation_script: string;

  /** Short scene title. */
  title?: string;

  /** The specific gap/pattern this story is trying to repair. */
  root_problem_focus: string;

  /** The target mental model the learner should leave with. */
  target_takeaway: string;

  /**
   * Visual semantic extraction from orientation_script. This is what keeps the
   * scene tied to the words instead of becoming a generic template.
   */
  visual_semantics: VisualStoryVisualSemantics;

  /** Ordered cinematic beats. Each beat maps to a specific script segment. */
  beats: VisualStoryBeat[];

  /** Scene-level defaults and rendering hints. */
  scene_defaults?: VisualStorySceneDefaults | null;

  /** Caption/script behavior. */
  script_display?: VisualStoryScriptDisplay | null;
};

export type VisualStorySceneFamily =
  | "motion_to_graph"
  | "mechanism"
  | "process_flow"
  | "compare_contrast"
  | "spatial_structure"
  | "quantity_change"
  | "cause_effect"
  | "classification_boundary"
  | "generic_concept_scene";

export type VisualStoryVisualSemantics = {
  /** Core objects/concepts that should appear visually. */
  key_entities: VisualStoryEntitySpec[];

  /** Important relationships the learner should see. */
  key_relationships: VisualStoryRelationshipSpec[];

  /** Important motions, mappings, or transformations. */
  key_transformations: VisualStoryTransformationSpec[];

  /**
   * Examples mentioned by the orientation script that can be visual echoes.
   * Do not force examples that are not in the orientation_script.
   */
  supporting_examples?: VisualStorySupportingExampleSpec[];

  /** Entity or relationship ids that should be emphasized. */
  emphasis_targets?: string[];
};

  id: string;

  semantic_role: string;

  visual_style?: {

    color_hint?: string;
    emphasis?: "low" | "medium" | "high";
    persistent?: boolean;
  } | null;

  /** Optional geometry hints. Avoid relying on raw generated code. */
  geometry?: Record<string, unknown> | null;
};
