export function seconds(value: number) {

export function stableUniversalSceneContractId(input: string) {
  let hash = 0;
  for (let index = 0; index < input.length; index += 1) {
    hash = (hash * 31 + input.charCodeAt(index)) >>> 0;
  }
  return `universal_scene_contract_${hash.toString(16).padStart(8, "0")}`;
}

function clamp(value: number, min: number, max: number) {
  return Math.min(max, Math.max(min, value));
}

function asRecord(value: unknown): Record<string, unknown> {
  return value && typeof value === "object" ? (value as Record<string, unknown>) : {};
}

function asText(value: unknown, fallback: string, maxLength = 240) {
  if (typeof value !== "string") return fallback;
  const trimmed = value.trim().replace(/\s+/g, " ");
  if (!trimmed) return fallback;
  return trimmed.slice(0, maxLength);
}

function asNumber(value: unknown, fallback: number, min: number, max: number) {
  const numberValue = typeof value === "number" ? value : Number(value);
  if (!Number.isFinite(numberValue)) return fallback;
  return clamp(numberValue, min, max);
}

function asInteger(value: unknown, fallback: number, min: number, max: number) {
  return Math.round(asNumber(value, fallback, min, max));
}

function asStringArray(value: unknown, fallback: string[], maxItems = 12, maxLength = 90) {
  if (!Array.isArray(value)) return fallback;
  const cleaned = value
    .map((item) => asText(item, "", maxLength))
    .filter(Boolean)
    .slice(0, maxItems);
  return cleaned.length ? cleaned : fallback;
}

function asTone(value: unknown, fallback: UniversalSceneTone): UniversalSceneTone {
  return typeof value === "string" && TONES.has(value as UniversalSceneTone)
    ? (value as UniversalSceneTone)
    : fallback;
}

function asElementKind(value: unknown, fallback: UniversalSceneElementKind): UniversalSceneElementKind {
  return typeof value === "string" && ELEMENT_KINDS.has(value as UniversalSceneElementKind)
    ? (value as UniversalSceneElementKind)
    : fallback;
}

function asActionKind(value: unknown, fallback: UniversalSceneActionKind): UniversalSceneActionKind {
  return typeof value === "string" && ACTION_KINDS.has(value as UniversalSceneActionKind)
    ? (value as UniversalSceneActionKind)
    : fallback;

function sanitizeId(value: unknown, fallback: string) {
  return asText(value, fallback, 60).replace(/[^a-z0-9_-]/gi, "_").replace(/_+/g, "_").toLowerCase();
}

function asPoint(value: unknown, fallback: UniversalScenePoint): UniversalScenePoint {
  if (Array.isArray(value) && value.length >= 2) {
    return [
      asNumber(value[0], fallback[0], SAFE_STAGE_X_MIN, SAFE_STAGE_X_MAX),
      asNumber(value[1], fallback[1], SAFE_STAGE_Y_MIN, SAFE_STAGE_Y_MAX),
    ];
  }

  if (value && typeof value === "object") {
    const record = value as Record<string, unknown>;
    return [
      asNumber(record.x, fallback[0], SAFE_STAGE_X_MIN, SAFE_STAGE_X_MAX),
      asNumber(record.y, fallback[1], SAFE_STAGE_Y_MIN, SAFE_STAGE_Y_MAX),
    ];
  }

  return [
    asNumber(fallback[0], 640, SAFE_STAGE_X_MIN, SAFE_STAGE_X_MAX),
    asNumber(fallback[1], 360, SAFE_STAGE_Y_MIN, SAFE_STAGE_Y_MAX),
  ];
}

function normalizePoints(value: unknown, fallback: UniversalScenePoint[]): UniversalScenePoint[] {
  if (!Array.isArray(value)) return fallback;
  const points = value.slice(0, 12).map((point, index) => asPoint(point, fallback[index] ?? [640, 360]));
  return points.length >= 3 ? points : fallback;
}


function sanitizePathD(value: unknown): string | undefined {
  const d = asText(value, "", 280);
  if (!d) return undefined;

  const numbers = Array.from(d.matchAll(/-?\d+(?:\.\d+)?/g)).map((match) => Number(match[0]));
  if (!numbers.length) return undefined;

  const hasClearlyOffCanvasNumber = numbers.some((number) => number < 0 || number > UNIVERSAL_SCENE_WIDTH);
  if (hasClearlyOffCanvasNumber) return undefined;

  return d;
}

function clampBoxCoordinate(args: {
  value: unknown;
  fallback: number;
  min: number;
  max: number;
  size: number;
}) {
  return asNumber(args.value, args.fallback, args.min, Math.max(args.min, args.max - args.size));
}

function normalizeElement(raw: unknown, index: number): UniversalSceneElement | null {
  const record = asRecord(raw);
  if (!Object.keys(record).length) return null;

  const kind = asElementKind(record.kind ?? record.type, index % 5 === 0 ? "card" : "rect");
  const id = sanitizeId(record.id, `element_${index + 1}`);
  const width = asNumber(record.width, kind === "text" ? 320 : kind === "card" ? 230 : 150, 12, 640);
  const height = asNumber(record.height, kind === "text" ? 70 : kind === "card" ? 120 : 90, 12, 380);
  const fallbackX = 190 + (index % 4) * 250;
  const fallbackY = 175 + Math.floor(index / 4) * 120;
  const usesTopLeft = kind === "card" || kind === "rect" || kind === "highlight";
  const x = usesTopLeft
    ? clampBoxCoordinate({ value: record.x, fallback: fallbackX, min: SAFE_STAGE_X_MIN, max: SAFE_STAGE_X_MAX, size: width })
    : asNumber(record.x, fallbackX, SAFE_STAGE_X_MIN, SAFE_STAGE_X_MAX);
  const y = usesTopLeft
    ? clampBoxCoordinate({ value: record.y, fallback: fallbackY, min: SAFE_STAGE_Y_MIN, max: SAFE_STAGE_Y_MAX, size: height })
    : asNumber(record.y, fallbackY, SAFE_STAGE_Y_MIN, SAFE_STAGE_Y_MAX);

  return {
    id,
    kind,
    label: asText(record.label, "", 80) || undefined,
    text: asText(record.text, "", 120) || undefined,
    icon: asText(record.icon, "", 38) || undefined,
    x,
    y,
    width,
    height,
    radius: asNumber(record.radius ?? record.r, kind === "circle" ? 42 : 24, 4, 120),
    rx: asNumber(record.rx, width / 2, 4, 220),
    ry: asNumber(record.ry, height / 2, 4, 160),
    points: normalizePoints(record.points, [
      [x, Math.max(SAFE_STAGE_Y_MIN, y - 60)],
      [Math.min(SAFE_STAGE_X_MAX, x + 90), y],
      [x, Math.min(SAFE_STAGE_Y_MAX, y + 60)],
      [Math.max(SAFE_STAGE_X_MIN, x - 90), y],
    ]),
    from: asPoint(record.from, [x, y]),
    to: asPoint(record.to, [Math.min(SAFE_STAGE_X_MAX, x + 180), y]),
    d: sanitizePathD(record.d),
    tone: asTone(record.tone, ["purple", "blue", "cyan", "green", "yellow", "orange", "pink"][index % 7] as UniversalSceneTone),
    opacity: asNumber(record.opacity, 1, 0, 1),
    rotation: asNumber(record.rotation, 0, -360, 360),
    scale: asNumber(record.scale, 1, 0.2, 3),
    layer: asInteger(record.layer, index, -20, 120),
  };
}

function normalizeElements(value: unknown, learnerSignal: string): UniversalSceneElement[] {
  if (Array.isArray(value)) {
    const cleaned = value
      .slice(0, 42)
      .map((item, index) => normalizeElement(item, index))
      .filter((item): item is UniversalSceneElement => Boolean(item));

    if (cleaned.length >= 3) return cleaned;
  }

  return buildFallbackUniversalSceneContract(learnerSignal).elements;
}

function normalizeAction(raw: unknown, index: number): UniversalSceneAction | null {
  const record = asRecord(raw);
  if (!Object.keys(record).length) return null;

  const to = asRecord(record.to);
  const targetId = sanitizeId(record.target_id ?? record.targetId ?? record.target, "");
  if (!targetId) return null;

  return {
    target_id: targetId,
    kind: asActionKind(record.kind ?? record.action, index % 3 === 0 ? "highlight" : "appear"),
    startFrame: asInteger(record.startFrame ?? record.start_frame, 0, 0, seconds(9)),
    endFrame: asInteger(record.endFrame ?? record.end_frame, seconds(2), 1, seconds(12)),
    to: Object.keys(to).length
      ? {
          x: to.x === undefined ? undefined : asNumber(to.x, 640, SAFE_STAGE_X_MIN, SAFE_STAGE_X_MAX),
          y: to.y === undefined ? undefined : asNumber(to.y, 360, SAFE_STAGE_Y_MIN, SAFE_STAGE_Y_MAX),
          opacity: to.opacity === undefined ? undefined : asNumber(to.opacity, 1, 0, 1),
          scale: to.scale === undefined ? undefined : asNumber(to.scale, 1, 0.2, 3),
          rotation: to.rotation === undefined ? undefined : asNumber(to.rotation, 0, -360, 360),
        }
      : undefined,
    label: asText(record.label, "", 80) || undefined,
  };
}

function normalizeScenes(value: unknown, elements: UniversalSceneElement[], learnerSignal: string): UniversalSceneBeat[] {
  if (Array.isArray(value)) {
    const allIds = new Set(elements.map((element) => element.id));
    const cleaned = value
      .slice(0, 7)
      .map((raw, index): UniversalSceneBeat | null => {
        const record = asRecord(raw);
        if (!Object.keys(record).length) return null;
        const visibleFallback = elements.slice(0, Math.min(elements.length, 8)).map((element) => element.id);
        const visible = asStringArray(record.visible_element_ids ?? record.visibleElements, visibleFallback, 24, 60).filter((id) => allIds.has(sanitizeId(id, id)) || allIds.has(id));
        const spotlight = asStringArray(record.spotlight_element_ids ?? record.spotlightElements, [], 8, 60).filter((id) => allIds.has(sanitizeId(id, id)) || allIds.has(id));
        const actions = Array.isArray(record.actions)
          ? record.actions
              .slice(0, 14)
              .map((item, actionIndex) => normalizeAction(item, actionIndex))
              .filter((action): action is UniversalSceneAction => action !== null && allIds.has(action.target_id))
          : [];

        return {
          id: sanitizeId(record.id, `scene_${index + 1}`),
          title: asText(record.title, index === 0 ? "Set up the idea" : `Scene ${index + 1}`, 72),
          caption: asText(record.caption, "Watch the pieces move so the hidden link becomes visible.", 190),
          durationInFrames: asInteger(record.durationInFrames ?? record.duration_in_frames ?? record.durationSeconds, seconds(4.5), seconds(2.5), seconds(8)),
          focus: asText(record.focus, "Focus on the part that changes.", 160),
          visible_element_ids: visible.length ? visible : visibleFallback,
          spotlight_element_ids: spotlight,
          actions,
        };
      })
      .filter((scene): scene is UniversalSceneBeat => Boolean(scene));

    if (cleaned.length >= 2) {
      const hasCheckpoint = cleaned.some((scene) => /check|try|explain|your/i.test(scene.title));
      if (!hasCheckpoint) {
        cleaned.push(buildDefaultCheckpointScene(elements));
      }
      return cleaned;
    }
  }

  return buildFallbackUniversalSceneContract(learnerSignal).scenes;
}

function buildDefaultCheckpointScene(elements: UniversalSceneElement[]): UniversalSceneBeat {
  return {
    id: "checkpoint",
    title: "Checkpoint",
    caption: "Name the one link that changed your understanding.",
    durationInFrames: seconds(4),
    focus: "Ask the learner to explain the cause/effect link.",
    visible_element_ids: elements.slice(0, Math.min(elements.length, 10)).map((element) => element.id),
    spotlight_element_ids: elements.slice(0, 2).map((element) => element.id),
    actions: elements.slice(0, 2).map((element, index) => ({
      target_id: element.id,
      kind: index === 0 ? "pulse" : "highlight",
      startFrame: 8,
      endFrame: seconds(3),
    })),
  };
}

function defaultSceneStyle(input: unknown): UniversalSceneContract["scene_style"] {
  if (typeof input === "string") {

      background: "deep_space",
      notes: ["Use simple original shapes.", "Make the hidden relationship visible."],

  const record = asRecord(input);
  const background = asText(record.background, "deep_space", 40);
  return {
    visual_language: asText(record.visual_language, "Clean animated educational diagram with glass labels, arrows, and gentle motion.", 160),
    background: background === "paper" || background === "chalkboard" || background === "clean_room" ? background : "deep_space",

  };
}

export function normalizeUniversalSceneContract(input: unknown, learnerSignalFallback?: string): UniversalSceneContract {
  const record = asRecord(input);
  const learnerSignal = asText(record.learner_signal ?? record.learnerSignal, learnerSignalFallback ?? "I am stuck on this idea.", 420);
  const elements = normalizeElements(record.elements, learnerSignal);
  const scenes = normalizeScenes(record.scenes, elements, learnerSignal);
  const checkpoint = asRecord(record.checkpoint);

  return {
    schema_version: "myway_universal_scene_contract_v1",
    contract_id: asText(record.contract_id, stableUniversalSceneContractId(`${learnerSignal}:${JSON.stringify(record).slice(0, 600)}`), 80),
    title: asText(record.title, "A visual explanation built from your message", 84),
    learner_signal: learnerSignal,
    diagnosis_guess: asText(record.diagnosis_guess, "The learner needs the hidden relationship turned into visible motion.", 240),
    learning_goal: asText(record.learning_goal, "Show the missing link with an animated scene and a checkpoint.", 240),
    director_plan: normalizeDirectorPlan(record.director_plan ?? record.directorPlan, learnerSignal),
    format: {
      width: UNIVERSAL_SCENE_WIDTH,
      height: UNIVERSAL_SCENE_HEIGHT,
      fps: UNIVERSAL_SCENE_FPS,
      aspect: "16:9_laptop",
    },
    renderer_strategy: makeRendererStrategy(),
    scene_style: defaultSceneStyle(record.scene_style),
    elements,
    scenes,
    checkpoint: {
      prompt: asText(checkpoint.prompt, "What changed in your understanding after watching the animation?", 160),
      expected_idea: asText(checkpoint.expected_idea, "The learner names the main cause/effect or structure highlighted by the animation.", 220),
    },
    safety_notes: [
      "The model outputs JSON director contracts and scene graphs, not executable code.",
      "MyWay validates element kinds, stage-safe coordinates, labels, scene length, and actions before rendering.",
      "The scene can be novel, but it must be built from safe vector primitives and stabilized layout rules.",
    ],
  };
}

export function getUniversalSceneDurationInFrames(contract: UniversalSceneContract) {
  return contract.scenes.reduce((sum, scene) => sum + scene.durationInFrames, 0);
}

export function findUniversalSceneForFrame(contract: UniversalSceneContract, frame: number) {
  let cursor = 0;

  for (let index = 0; index < contract.scenes.length; index += 1) {
    const scene = contract.scenes[index]!;
    const startFrame = cursor;
    const endFrame = startFrame + scene.durationInFrames;

    if (frame <= endFrame || index === contract.scenes.length - 1) {
      return {
        scene,
        sceneIndex: index,
        startFrame,
        endFrame,
        progress: Math.max(0, Math.min(1, (frame - startFrame) / Math.max(1, scene.durationInFrames))),
      };
    }

    cursor = endFrame;
  }

  const fallback = contract.scenes[0] ?? buildDefaultCheckpointScene(contract.elements);
  return {
    scene: fallback,
    sceneIndex: 0,
    startFrame: 0,
    endFrame: fallback.durationInFrames,
    progress: 0,
  };
}

function baseContract(args: {
  learnerSignal: string;
  title: string;
  diagnosis: string;
  goal: string;
  elements: UniversalSceneElement[];
  scenes: UniversalSceneBeat[];
  checkpoint: UniversalSceneContract["checkpoint"];
  style?: Partial<UniversalSceneContract["scene_style"]>;
}): UniversalSceneContract {
  return normalizeUniversalSceneContract({
    schema_version: "myway_universal_scene_contract_v1",
    contract_id: stableUniversalSceneContractId(args.learnerSignal),
    title: args.title,
    learner_signal: args.learnerSignal,
    diagnosis_guess: args.diagnosis,
    learning_goal: args.goal,
    scene_style: {
      visual_language: args.style?.visual_language ?? "Clean animated educational diagram with original vector shapes.",
      background: args.style?.background ?? "deep_space",
      notes: args.style?.notes ?? ["Fallback generated by MyWay.", "Use safe primitives."],
    },
    elements: args.elements,
    scenes: args.scenes,
    checkpoint: args.checkpoint,
  }, args.learnerSignal);
}
