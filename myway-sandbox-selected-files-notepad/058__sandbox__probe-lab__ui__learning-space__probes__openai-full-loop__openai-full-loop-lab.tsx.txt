function clamp01(value: number) {
  if (!Number.isFinite(value)) return 0;
  return Math.max(0, Math.min(1, value));
}

function round3(value: number) {
  return Math.round(value * 1000) / 1000;

function asRecord(value: unknown): Record<string, unknown> {
  return value && typeof value === "object" && !Array.isArray(value) ? (value as Record<string, unknown>) : {};

function stableIdFromLabel(label: string) {
  const slug = label
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "_")
    .replace(/^_+|_+$/g, "")

  return `topic_${slug || "untitled"}`;
}

function stringifyAttempt(attempt: ProbeAnswerDraft | null) {
  if (!attempt) return "";
  if (attempt.text_response?.trim()) return attempt.text_response.trim();
  if (attempt.audio_response_transcript?.trim()) return attempt.audio_response_transcript.trim();
  if (attempt.selected_option_id) return attempt.selected_option_id;
  if (attempt.selected_option_ids?.length) return attempt.selected_option_ids.join(", ");
  if (attempt.ordered_item_ids?.length) return attempt.ordered_item_ids.join(" -> ");
  if (attempt.placements && Object.keys(attempt.placements).length > 0) return JSON.stringify(attempt.placements);
  if (typeof attempt.numeric_response === "number") return String(attempt.numeric_response);
  if (attempt.graph_features?.length) return attempt.graph_features.join("\n");
  if (typeof attempt.selected_click_seconds === "number") return String(attempt.selected_click_seconds);
  return "";
}

function normalizeGraphFunctions(value: unknown): ProbeGraphFunctionDraft[] {
  if (!Array.isArray(value)) return [];
  const functions: ProbeGraphFunctionDraft[] = [];
  for (const item of value.slice(0, 4)) {
    const r = asRecord(item);
    const expression = safeString(r.expression);
    if (!expression) continue;
    functions.push({
      id: safeString(r.id, `fn_${functions.length + 1}`),
      label: safeString(r.label, `f${functions.length + 1}`),
      expression,
      enabled: typeof r.enabled === "boolean" ? r.enabled : true,
    });
  }
  return functions;
}

function normalizeGraphParameters(value: unknown): ProbeGraphParameterDraft[] {
  if (!Array.isArray(value)) return [];
  const parameters: ProbeGraphParameterDraft[] = [];
  for (const item of value.slice(0, 5)) {
    const r = asRecord(item);
    const name = safeString(r.name).replace(/[^a-zA-Z]/g, "").slice(0, 1);
    const valueNumber = typeof r.value === "number" && Number.isFinite(r.value) ? r.value : null;
    if (!name || valueNumber === null) continue;
    parameters.push({
      name,
      value: valueNumber,
      min: typeof r.min === "number" ? r.min : -5,
      max: typeof r.max === "number" ? r.max : 5,
      step: typeof r.step === "number" ? r.step : 0.1,
    });
  }
  return parameters;
}

function seedDraftForProbe(probe: EngineRenderableProbe): ProbeAnswerDraft {
  const draft = createEmptyProbeAnswerDraft(probe.expected_attempt_type);
  const rendererParams = asRecord(probe.renderer_params);
  const graph = asRecord(rendererParams.graph);
  const mode = graph.mode === "3d" ? "3d" : graph.mode === "2d" ? "2d" : null;

  if (probe.probe_type === "graph_relationship" || probe.expected_attempt_type === "graph") {
    return {
      ...draft,
      graph_mode: mode ?? draft.graph_mode,
      graph_functions: normalizeGraphFunctions(graph.functions),
      graph_parameters: normalizeGraphParameters(graph.parameters),
      graph_window: asRecord(graph.window) as ProbeAnswerDraft["graph_window"],
      graph_3d_window: asRecord(graph.window3d) as ProbeAnswerDraft["graph_3d_window"],
      graph_3d_view: asRecord(graph.view3d) as ProbeAnswerDraft["graph_3d_view"],
      graph_surface_expression: safeString(graph.surface_expression, draft.graph_surface_expression ?? ""),
      graph_notes: safeString(graph.prompt_notes, draft.graph_notes ?? ""),
    };
  }

  return draft;
}

function misconceptionLabels(probe: EngineRenderableProbe) {
  return (probe.misconception_markers ?? [])
    .map((marker) => marker.label || marker.misconception_id)
    .filter((item): item is string => typeof item === "string" && item.trim().length > 0)
    .slice(0, 8);
}

function applyDelta(topic: SandboxTopicState, delta?: MetricDelta | null) {
  if (!delta) return topic;
  return {
    ...topic,
    confusion: round3(clamp01(topic.confusion + delta.confusion_delta)),
    insight: round3(clamp01(topic.insight + delta.insight_delta)),
    mastery: round3(clamp01(topic.mastery + delta.mastery_delta)),
    evidence_count: Math.max(0, topic.evidence_count + Math.max(0, Math.floor(delta.evidence_count_delta))),
    verification_needed: delta.verification_needed,
    updated_at: nowIso(),
  };
}

function topicFromGenerate(args: { previous?: SandboxTopicState; output: GenerateOutput; fallbackTopicId?: string | null }) {
  const diagnosis = asRecord(args.output.diagnosis);
  const topicDecision = args.output.topic_decision;
  const label = topicDecision?.topic_label || safeString(diagnosis.target_topic_label, "Auto-detected topic");
  const topicId = topicDecision?.topic_id || args.fallbackTopicId || stableIdFromLabel(label);
  const base = args.previous ?? createTopic({ topicId, topicLabel: label, diagnosis: safeString(diagnosis.diagnosis, "unknown"), rootProblem: safeString(diagnosis.root_problem) });
  const withDelta = applyDelta(base, args.output.metric_delta ?? null);
  return {
    ...withDelta,
    topic_id: base.topic_id || topicId,
    topic_label: label,
    message_count: base.message_count + 1,
    active_diagnosis: safeString(diagnosis.diagnosis, base.active_diagnosis ?? "unknown"),
    diagnosis_belief: round3(clamp01(Math.max(base.diagnosis_belief, Number(diagnosis.diagnosis_confidence ?? 0.5) * 0.82))),
    diagnosis_confidence: round3(clamp01(Number(diagnosis.diagnosis_confidence ?? base.diagnosis_confidence))),
    diagnosis_status: withDelta.verification_needed ? "active" : withDelta.mastery > 0.72 ? "weakening" : "active",
    root_problem: safeString(diagnosis.root_problem, base.root_problem ?? ""),
    misconception_patterns: [...new Set([...base.misconception_patterns, ...misconceptionLabels(args.output.probe)])].slice(0, 12),
    latest_probe_type: args.output.probe.probe_type,
    updated_at: nowIso(),
  } satisfies SandboxTopicState;
}

function topicFromEvaluation(topic: SandboxTopicState, evaluation: EvaluationOutput, probe: EngineRenderableProbe): SandboxTopicState {
  const next = applyDelta(topic, evaluation.metric_delta ?? null);
  const hits = (evaluation.misconception_hits ?? [])
    .map((hit) => hit.label || hit.misconception_id)
    .filter((item): item is string => typeof item === "string" && item.trim().length > 0);
  const evidence = asRecord(evaluation.understanding_evidence);
  return {
    ...next,
    evidence_count: Math.max(next.evidence_count, topic.evidence_count + 1),
    verification_needed: Boolean(evidence.needs_verification_probe ?? next.verification_needed),
    diagnosis_status: evaluation.correctness >= 0.88 && !Boolean(evidence.needs_verification_probe) ? "weakening" : "active",
    latest_probe_type: probe.probe_type,
    misconception_patterns: [...new Set([...next.misconception_patterns, ...hits])].slice(0, 12),
    updated_at: nowIso(),
  };
}

function probeEvidenceContext(probe: EngineRenderableProbe | null) {
  if (!probe) return null;
  const selectedOptionProbe = probe.expected_attempt_type === "single_choice" || probe.expected_attempt_type === "multi_choice";
  const placementProbe = probe.expected_attempt_type === "drag_drop_placements" || probe.expected_attempt_type === "ordered_items";
  const graphProbe = probe.expected_attempt_type === "graph";
  const textProbe = probe.expected_attempt_type === "text";

  return {
    probe_type: probe.probe_type,
    expected_attempt_type: probe.expected_attempt_type,
    evidence_bandwidth: selectedOptionProbe ? "low" : placementProbe || graphProbe ? "medium" : textProbe ? "high" : "medium",
    can_measure: selectedOptionProbe
      ? ["recognition", "basic discrimination"]
      : placementProbe
        ? ["classification", "mapping", "pattern separation"]
        : graphProbe
          ? ["visual model", "representation mapping", "feature noticing"]
          : textProbe
            ? ["explanation", "reasoning", "transfer depth"]
            : ["task performance"],
    cannot_measure: selectedOptionProbe
      ? ["explanation", "reasoning chain", "transfer depth"]
      : [],
    evaluator_rule: selectedOptionProbe
      ? "Do not criticize the learner for not explaining. The probe did not ask for explanation. Treat a correct selected option as recognition evidence and cap evidence strength unless another response field shows reasoning."
      : "Judge only what the delivered probe actually asked the learner to do.",
  };
}

function messageUnderstandingLabel(output: GenerateOutput | null) {
  const score = output?.message_understanding?.confidence;
  if (typeof score !== "number") return "unknown";
  if (score < 0.45) return "low";
  if (score < 0.72) return "medium";
  return "high";
}

    () => profile.interests_text.split(",").map((item) => item.trim()).filter(Boolean).slice(0, 8),
    [profile.interests_text],
  );

  function upsertTopic(nextTopic: SandboxTopicState, turn?: SandboxTurn) {
    setTopics((current) => ({
      ...current,
      [nextTopic.topic_id]: {
        ...nextTopic,
        history: turn ? [...nextTopic.history, turn].slice(-20) : nextTopic.history,
      },
    }));
    setActiveTopicId(nextTopic.topic_id);
    if (turn) setHistory((current) => [turn, ...current].slice(0, 60));
  }

  function currentPayload() {
    return {
      active_topic_id: activeTopicId,
      existing_topics: topicList.map(topicSummary),
      current_topic_state: activeTopic ? topicSummary(activeTopic) : null,
      recent_history: history.slice(0, 10),
      personalization_context: {
        schema_version: "sandbox_personalization_profile_v31",
        bridge_level: profile.bridge_level,
        language_policy: { jargon_level: profile.jargon_level },
        preferred_style: profile.preferred_style,
        teaching_recipe: {
          preferred_styles: profile.teaching_recipe.preferred_styles_text
            .split(",")
            .map((item) => item.trim())
            .filter(Boolean)
            .slice(0, 10),
          avoid_styles: profile.teaching_recipe.avoid_styles_text
            .split(",")
            .map((item) => item.trim())
            .filter(Boolean)
            .slice(0, 10),
          explanation_shape: profile.teaching_recipe.explanation_shape,
          feedback_style: profile.teaching_recipe.feedback_style,
          challenge_level: profile.teaching_recipe.challenge_level,
          pacing: profile.teaching_recipe.pacing,
          verification_preference: profile.teaching_recipe.verification_preference,
        },
        user_interests: interests,
        interests: interests.map((domain) => ({
          domain,
          usefulness: 0.5,
          use_when: profile.interest_use_policy,
          avoid_when: "Avoid using this interest when it does not clarify the target concept or would make the probe feel gimmicky.",
          recent_use_count: 0,
        })),
        interest_use_policy: profile.interest_use_policy,
        profile_summary: profile.profile_summary,
        learner_notes: profile.learner_notes,
        personalization_events: profile.personalization_events.slice(-8),
      },
    };
  }

  async function generateProbe(reason: "message" | "followup_from_attempt" = "message") {
    setIsGenerating(true);
    setGenerateResult(null);
    setEvaluationResult(reason === "message" ? null : evaluationResult);
    setImageResult(null);
    setDraft(null);
    setSubmittedAttempt(null);

    try {
      const response = await fetch("/api/sandbox/probe-lab/openai-full-loop/generate", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          learner_message:
            reason === "followup_from_attempt"
              ? `Generate the next best follow-up probe from the latest evaluated attempt. Original learner signal: ${learnerMessage}`
              : learnerMessage,
          target_topic_label: topicLabel,
          topic_mode: reason === "followup_from_attempt" ? "continue_active" : topicMode,
          bridge_level: profile.bridge_level,
          jargon_level: profile.jargon_level,
          preferred_style: profile.preferred_style,
          user_interests: interests,
          profile_summary: profile.profile_summary,
          previous_evaluation: evaluationResult?.output ?? null,
          turn_mode: output?.turn_mode ?? "probe_only",
          generate_reason: reason,
          model,
          ...currentPayload(),
        }),
      });

      const data = (await response.json()) as ApiResult<GenerateOutput>;
      setGenerateResult(data);

      if (data.output?.probe) {
        setDraft(seedDraftForProbe(data.output.probe));
      }

      if (data.output) {
        const decision = data.output.topic_decision;
        const diagnosis = asRecord(data.output.diagnosis);
        const targetId =
          decision?.action === "continue_existing_topic" && activeTopic
            ? activeTopic.topic_id
            : decision?.action === "switch_to_existing_topic" && decision.topic_id && topics[decision.topic_id]
              ? decision.topic_id
              : decision?.action === "create_new_topic"
                ? decision.topic_id || stableIdFromLabel(decision.topic_label)
                : activeTopic?.topic_id ?? decision?.topic_id ?? stableIdFromLabel(decision?.topic_label || safeString(diagnosis.target_topic_label, "New topic"));
        const previous = topics[targetId] ?? null;
        const nextTopic = topicFromGenerate({ previous: previous ?? undefined, output: data.output, fallbackTopicId: targetId });
        const turn: SandboxTurn = {
          id: `turn_${Date.now()}`,
          kind: reason === "followup_from_attempt" ? "followup_probe" : "generated_probe",
          topic_id: nextTopic.topic_id,
          topic_label: nextTopic.topic_label,
          created_at: nowIso(),
          summary: `${reason === "followup_from_attempt" ? "Follow-up" : "Generated"} ${data.output.probe.probe_type} probe · ${data.output.rationale.what_it_measures}`,
          probe_type: data.output.probe.probe_type,
          diagnosis: safeString(diagnosis.diagnosis, null as unknown as string) || null,
          correctness: null,
          metric_delta: data.output.metric_delta ?? null,
        };
        upsertTopic(nextTopic, turn);
      }
    } catch (error) {
      setGenerateResult({ ok: false, error: error instanceof Error ? error.message : "Unknown generate error" });
    } finally {
      setIsGenerating(false);
    }
  }

  async function evaluateAttempt() {
    if (!probe || !activeDraft) return;
    setIsEvaluating(true);
    setEvaluationResult(null);

    try {
      const response = await fetch("/api/sandbox/probe-lab/openai-full-loop/evaluate", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({
          model,
          learner_message: learnerMessage,
          diagnosis: output?.diagnosis ?? null,
          probe,
          attempt: activeDraft,
          probe_evidence_context: probeEvidenceContext(probe),
          topic_state: activeTopic ? topicSummary(activeTopic) : null,
          ...currentPayload(),
        }),
      });

      const data = (await response.json()) as ApiResult<EvaluationOutput>;
      setEvaluationResult(data);

      const evaluationOutput = data.output;
      if (!evaluationOutput) {
        return;
      }

      if (data.output && activeTopic) {
        const nextTopic = topicFromEvaluation(activeTopic, data.output, probe);
        const turn: SandboxTurn = {
          id: `turn_${Date.now()}`,
          kind: "attempt_evaluation",
          topic_id: nextTopic.topic_id,
          topic_label: nextTopic.topic_label,
          created_at: nowIso(),
          summary: String(evaluationOutput.correctness_summary ?? "Attempt evaluated."),
          probe_type: probe.probe_type,
          diagnosis: nextTopic.active_diagnosis,
          correctness: evaluationOutput.correctness,
          metric_delta: evaluationOutput.metric_delta ?? null,
        };
        upsertTopic(nextTopic, turn);

        if (evaluationOutput.personalization_delta) {
          setProfile((current) => ({
            ...current,
            personalization_events: [...current.personalization_events, evaluationOutput.personalization_delta].slice(-20),
            profile_summary: `${current.profile_summary}\nLatest evidence: ${(evaluationOutput.learner_feedback as Record<string, unknown> | undefined)?.suggested_next_step ?? String(evaluationOutput.correctness_summary ?? "Attempt evaluated.")}`.slice(0, 1200),
          }));
        }
      }
    } catch (error) {
      setEvaluationResult({ ok: false, error: error instanceof Error ? error.message : "Unknown evaluate error" });
    } finally {
      setIsEvaluating(false);
    }
  }

  async function generateImage() {
    if (!output?.image_prompt) return;
    setIsGeneratingImage(true);
    setImageResult(null);

    try {
      const response = await fetch("/api/sandbox/probe-lab/openai-full-loop/generate-image", {

        headers: { "content-type": "application/json" },

      });

    } catch (error) {
      setImageResult({ ok: false, error: error instanceof Error ? error.message : "Unknown image error" });
    } finally {
      setIsGeneratingImage(false);
    }
  }

  async function buildRelationships() {
    setIsBuildingRelationships(true);
    setRelationshipResult(null);
    try {
      const response = await fetch("/api/sandbox/probe-lab/openai-full-loop/relationships", {

        headers: { "content-type": "application/json" },

    } catch (error) {
      setRelationshipResult({ ok: false, error: error instanceof Error ? error.message : "Unknown relationship error" });
    } finally {
      setIsBuildingRelationships(false);
    }
  }

              <p className="text-xs font-semibold uppercase tracking-[0.22em] text-amber-100/75">Clarification needed before teaching/probing</p>
              <h3 className="mt-2 text-xl font-semibold text-white">{output.clarification.question}</h3>
              <p className="mt-3 text-sm leading-6 text-amber-50/80">{output.clarification.reason}</p>
              {output.clarification.options?.length ? (
                <div className="mt-4 grid gap-2">
                  {output.clarification.options.map((option) => (
                    <button key={option} type="button" onClick={() => setLearnerMessage(`${learnerMessage}\n\nClarification answer: ${option}`)} className="rounded-2xl border border-amber-100/20 bg-black/20 px-4 py-3 text-left text-sm text-amber-50 hover:bg-amber-200/10">
                      {option}
                    </button>
                  ))}
                </div>
              ) : null}
            </section>
          ) : null}

          {output?.orientation ? (
            <section className="rounded-[2rem] border border-cyan-100/20 bg-cyan-300/10 p-5 shadow-2xl backdrop-blur-xl">
              <p className="text-xs font-semibold uppercase tracking-[0.22em] text-cyan-100/75">Orientation before the check</p>
              <h3 className="mt-2 text-xl font-semibold text-white">{output.orientation.title}</h3>
              <p className="mt-3 whitespace-pre-wrap text-sm leading-6 text-cyan-50/88">{output.orientation.body}</p>
              {output.orientation.visual_scene ? <p className="mt-3 rounded-2xl border border-white/10 bg-black/20 p-3 text-sm leading-6 text-zinc-200">{output.orientation.visual_scene}</p> : null}
              {output.orientation.key_points?.length ? (
                <ul className="mt-3 grid gap-2 text-sm leading-6 text-zinc-200">
                  {output.orientation.key_points.map((point) => <li key={point}>• {point}</li>)}
                </ul>
              ) : null}
              {output.orientation.bridge_note ? <p className="mt-3 text-xs leading-5 text-zinc-400">{output.orientation.bridge_note}</p> : null}
            </section>
          ) : null}

          {output?.interest_discovery ? (
            <section className="rounded-[2rem] border border-purple-100/20 bg-purple-300/10 p-5 shadow-2xl backdrop-blur-xl">
              <p className="text-xs font-semibold uppercase tracking-[0.22em] text-purple-100/75">Interest discovery</p>
              {output.interest_discovery.should_ask ? <p className="mt-2 text-sm leading-6 text-purple-50/85">{output.interest_discovery.question}</p> : null}
              {output.interest_discovery.inferred_candidates?.length ? (
                <div className="mt-3 grid gap-2">
                  {output.interest_discovery.inferred_candidates.map((candidate) => (
                    <div key={candidate.interest} className="rounded-2xl border border-white/10 bg-black/20 p-3 text-sm leading-6 text-zinc-200">
                      <b>{candidate.interest}</b> · confidence {candidate.confidence.toFixed(2)}
                      <p className="text-zinc-400">Use when: {candidate.use_when}</p>
                      <p className="text-zinc-500">Avoid when: {candidate.avoid_when}</p>
                    </div>
                  ))}
                </div>
              ) : null}
            </section>
          ) : null}

          {imageResult?.image_url ? (
            <section className="rounded-[2rem] border border-white/10 bg-white/[0.055] p-5 shadow-2xl backdrop-blur-xl">
              <p className="text-xs font-semibold uppercase tracking-[0.22em] text-zinc-300/70">Generated image stimulus</p>
              <img src={imageResult.image_url} alt="Generated probe stimulus" className="mt-4 w-full max-w-4xl rounded-3xl border border-white/10 object-cover shadow-2xl" />
            </section>
          ) : imageResult?.error ? <div className="rounded-3xl border border-red-200/20 bg-red-500/10 p-4 text-sm leading-6 text-red-100">{imageResult.error}</div> : null}

          {visibleProbe ? (
            <section className="overflow-hidden rounded-[2rem] border border-white/10 bg-black/[0.22] p-4 shadow-2xl backdrop-blur-xl">
              <ProbeRenderer
                key={`${visibleProbe.probe_type}-${visibleProbe.expected_attempt_type}-${visibleProbe.prompt.task}`}
                probe={visibleProbe}
                initialDraft={activeDraft ?? undefined}
                showDebug
                onDraftChange={(nextDraft) => { setDraft(nextDraft); setSubmittedAttempt(null); }}
                onSubmit={(payload) => { setSubmittedAttempt(payload); setDraft(payload.attempt); }}
              />
            </section>
          ) : null}
        </div>
      </section>

      <section className="grid gap-5 xl:grid-cols-[minmax(0,1fr)_minmax(0,1fr)]">
        <div className="rounded-[2rem] border border-white/10 bg-white/[0.045] p-5 shadow-2xl backdrop-blur-xl">
          <div className="flex items-center justify-between gap-3">
            <div>
              <p className="text-xs font-semibold uppercase tracking-[0.2em] text-zinc-300/70">Topics</p>
              <p className="mt-1 text-sm text-zinc-400">Switch topics without losing progress.</p>
            </div>
            <button type="button" onClick={buildRelationships} disabled={isBuildingRelationships || topicList.length < 2} className="rounded-2xl border border-white/10 bg-white/[0.07] px-4 py-2 text-sm font-semibold text-zinc-100 hover:bg-white/[0.11] disabled:opacity-45">
              {isBuildingRelationships ? "Building..." : "Build relationships"}
            </button>
          </div>
          <div className="mt-4 grid gap-2">
            {topicList.length ? topicList.map((topic) => (
              <button key={topic.topic_id} type="button" onClick={() => setActiveTopicId(topic.topic_id)} className={`rounded-2xl border p-3 text-left transition ${topic.topic_id === activeTopicId ? "border-emerald-200/35 bg-emerald-200/12" : "border-white/10 bg-black/15 hover:bg-white/[0.075]"}`}>
                <div className="flex items-center justify-between gap-3"><p className="font-semibold text-white">{topic.topic_label}</p><span className="text-xs text-zinc-400">{topic.evidence_count} evidence</span></div>
                <p className="mt-1 text-xs leading-5 text-zinc-400">{topic.active_diagnosis ?? "unknown"} · confusion {topic.confusion.toFixed(2)} · insight {topic.insight.toFixed(2)} · mastery {topic.mastery.toFixed(2)}</p>
              </button>
            )) : <p className="text-sm text-zinc-400">No topics yet.</p>}
          </div>
        </div>

        <div className="rounded-[2rem] border border-white/10 bg-white/[0.045] p-5 shadow-2xl backdrop-blur-xl">
          <p className="text-xs font-semibold uppercase tracking-[0.2em] text-zinc-300/70">Relationship preview</p>
          <p className="mt-2 text-sm leading-6 text-zinc-400">Shared diagnosis links are only between different topics. Continuation of one topic should update the same topic, not create a relationship.</p>
          <div className="mt-4 grid gap-3">
            {relationshipResult?.result?.relationships?.length ? relationshipResult.result.relationships.map((relationship) => (
              <div key={relationship.relationship_id} className="rounded-2xl border border-white/10 bg-black/20 p-3">
                <div className="flex flex-wrap items-center gap-2"><span className="rounded-full border border-white/10 bg-white/10 px-2.5 py-1 text-[11px] font-semibold uppercase tracking-[0.1em] text-zinc-200">{relationship.relationship_type}</span><span className="text-xs text-zinc-400">strength {relationship.strength.toFixed(2)} · confidence {relationship.confidence.toFixed(2)}</span></div>
                <p className="mt-2 text-sm leading-6 text-zinc-300">{relationship.evidence_summary ?? "No summary."}</p>
                {relationship.reasons?.length ? <p className="mt-2 text-xs leading-5 text-zinc-500">{relationship.reasons.slice(0, 5).join(" · ")}</p> : null}
              </div>
            )) : <p className="text-sm text-zinc-400">Build relationships after at least two topics exist.</p>}
            {relationshipResult?.error ? <div className="rounded-2xl border border-red-200/20 bg-red-500/10 p-3 text-sm text-red-100">{relationshipResult.error}</div> : null}
          </div>
        </div>
      </section>

      <section className="grid gap-4 xl:grid-cols-2">
<JsonPanel title="Message understanding / orientation" value={{ message_understanding: output?.message_understanding, turn_mode: output?.turn_mode, clarification: output?.clarification, orientation: output?.orientation, interest_discovery: output?.interest_discovery }} />
        <JsonPanel title="Diagnosis + probe output" value={output} />
        <JsonPanel title="Current attempt" value={{ submitted_attempt: submittedAttempt?.attempt ?? null, active_draft: activeDraft, response_summary: attemptPreview || null }} />
        <JsonPanel title="Evaluation output" value={evaluationResult?.output ?? evaluationResult} />
        <JsonPanel title="Relationship graph output" value={relationshipResult} open={false} />
        <JsonPanel title="Sandbox topics" value={topicList.map(topicSummary)} open={false} />
        <JsonPanel title="History" value={history} open={false} />
      </section>
    </section>
  );
}
