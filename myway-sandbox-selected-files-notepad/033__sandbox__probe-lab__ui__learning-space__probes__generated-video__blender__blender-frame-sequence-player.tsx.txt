"use client";

import { useEffect, useMemo, useState } from "react";

function formatTime(seconds: number) {
  const safe = Math.max(0, seconds);
  return `${Math.floor(safe / 60)}:${String(Math.floor(safe % 60)).padStart(2, "0")}`;
}

export function BlenderFrameSequencePlayer({
  frameUrls,
  fps = 12,
  title = "Blender render",
  status,
}: {
  frameUrls: string[];
  fps?: number;
  title?: string;
  status?: string | null;
}) {
  const [frameIndex, setFrameIndex] = useState(0);
  const [isPlaying, setIsPlaying] = useState(false);
  const safeFps = Math.max(1, Math.min(30, fps));
  const frameCount = frameUrls.length;
  const currentFrame = frameUrls[Math.min(frameIndex, Math.max(0, frameCount - 1))];
  const durationSeconds = useMemo(() => frameCount / safeFps, [frameCount, safeFps]);

  useEffect(() => {
    setFrameIndex(0);
    setIsPlaying(false);
  }, [frameUrls]);

  useEffect(() => {
    if (!isPlaying || frameCount <= 1) return;

    const interval = window.setInterval(() => {
      setFrameIndex((value) => {
        const next = value + 1;
        if (next >= frameCount) {
          setIsPlaying(false);
          return frameCount - 1;
        }
        return next;
      });
    }, 1000 / safeFps);

    return () => window.clearInterval(interval);
  }, [frameCount, isPlaying, safeFps]);

  if (!frameCount || !currentFrame) {
    return (
      <section
        style={{
          border: "1px solid rgba(255,255,255,0.12)",
          borderRadius: "24px",
          background: "rgba(0,0,0,0.22)",
          padding: "1rem",
          color: "rgba(255,255,255,0.76)",
        }}
      >
        No Blender frames rendered yet.
      </section>
    );
  }

  return (
    <section style={{ display: "grid", gap: "0.8rem" }}>
      <div
        style={{
          border: "1px solid rgba(255,255,255,0.12)",
          borderRadius: "28px",
          background: "linear-gradient(145deg, rgba(5,5,16,0.98), rgba(39,13,64,0.86))",
          overflow: "hidden",
          boxShadow: "0 24px 80px rgba(0,0,0,0.34)",
        }}
      >
        <div style={{ position: "relative", aspectRatio: "16 / 9" }}>
          {/* eslint-disable-next-line @next/next/no-img-element */}
          <img
            src={currentFrame}
            alt={`${title} frame ${frameIndex + 1}`}
            style={{ width: "100%", height: "100%", objectFit: "contain", display: "block", background: "#080011" }}
          />

          <div
            style={{
              position: "absolute",
              left: "1rem",
              top: "1rem",
              border: "1px solid rgba(255,255,255,0.14)",
              borderRadius: "999px",
              background: "rgba(0,0,0,0.42)",
              padding: "0.48rem 0.72rem",
              color: "white",
              fontSize: "0.78rem",
              fontWeight: 900,
              backdropFilter: "blur(10px)",
            }}
          >
            Blender frame sequence · {frameIndex + 1}/{frameCount}
          </div>
        </div>
      </div>

      <div style={{ display: "grid", gap: "0.6rem" }}>
        <div style={{ display: "flex", alignItems: "center", gap: "0.7rem" }}>
          <button
            type="button"
            onClick={() => {
              if (frameIndex >= frameCount - 1) setFrameIndex(0);
              setIsPlaying((value) => !value);
            }}
            style={{
              border: "1px solid rgba(221,214,254,0.38)",
              borderRadius: "999px",
              background: "rgba(221,214,254,0.16)",
              color: "white",
              padding: "0.58rem 0.88rem",
              fontWeight: 900,
              cursor: "pointer",
              minWidth: "5.8rem",
            }}
          >
            {isPlaying ? "Pause" : frameIndex >= frameCount - 1 ? "Replay" : "Play"}
          </button>

          <input
            aria-label="Blender frame timeline"
            type="range"
            min={0}
            max={Math.max(0, frameCount - 1)}
            step={1}
            value={frameIndex}
            onChange={(event) => {
              setFrameIndex(Number(event.target.value));
              setIsPlaying(false);
            }}
            style={{ width: "100%", accentColor: "#c4b5fd" }}
          />

          <span style={{ color: "rgba(255,255,255,0.72)", fontWeight: 850, fontSize: "0.8rem", whiteSpace: "nowrap" }}>
            {formatTime(frameIndex / safeFps)} / {formatTime(durationSeconds)}
          </span>
        </div>

        {status ? <p style={{ margin: 0, color: "rgba(255,255,255,0.66)", fontSize: "0.84rem", lineHeight: 1.45 }}>{status}</p> : null}
      </div>
    </section>
  );
}
