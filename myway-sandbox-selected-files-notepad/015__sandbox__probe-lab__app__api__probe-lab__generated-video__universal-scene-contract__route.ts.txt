function stripCodeFence(value: string) {
  const trimmed = value.trim();
  if (!trimmed.startsWith("```")) return trimmed;
  return trimmed
    .replace(/^```(?:json)?\s*/i, "")
    .replace(/\s*```$/i, "")
    .trim();
}

function safeParseJson(content: string) {
  const stripped = stripCodeFence(content);
  try {
    return JSON.parse(stripped) as unknown;
  } catch (error) {
    const firstBrace = stripped.indexOf("{");
    const lastBrace = stripped.lastIndexOf("}");
    if (firstBrace >= 0 && lastBrace > firstBrace) {
      return JSON.parse(stripped.slice(firstBrace, lastBrace + 1)) as unknown;
    }
    throw error;
  }
}

function contentToString(content: unknown) {
  if (typeof content === "string") return content;

  if (Array.isArray(content)) {
    return content
      .map((part) => {
        if (typeof part === "string") return part;
        if (part && typeof part === "object") {
          const record = part as Record<string, unknown>;
          if (typeof record.text === "string") return record.text;
          if (typeof record.content === "string") return record.content;
        }
        return "";
      })
      .join("")
      .trim();
  }

  return "";
}

async function fetchWithTimeout(url: string, init: RequestInit, timeoutMs: number) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), timeoutMs);

  try {
    return await fetch(url, {
      ...init,
      signal: controller.signal,
    });
  } finally {
    clearTimeout(timeout);
  }
}

function asRecord(value: unknown): Record<string, unknown> {
  return value && typeof value === "object" ? (value as Record<string, unknown>) : {};
}

function buildUniversalSceneFromDirector(args: {
  draftRecord: Record<string, unknown>;
  directorContract: MyWayVideoDirectorContract;
  context: VideoDirectorRequestContext;
  provider: UniversalSceneProvider;
  model: string;
}): UniversalSceneContract {
  const universalDraft =
    args.draftRecord.universal_scene_contract ??
    args.draftRecord.universalSceneContract ??
    args.draftRecord.universal_scene ??
    args.draftRecord.scene_graph ??
    args.draftRecord.sceneGraph ??
    args.draftRecord;

  const universalRecord = asRecord(universalDraft);
  const director = args.directorContract;

  return normalizeUniversalSceneContract(
    {
      ...universalRecord,
      schema_version: "myway_universal_scene_contract_v1",
      contract_id:
        typeof universalRecord.contract_id === "string" && universalRecord.contract_id.trim()
          ? universalRecord.contract_id
          : stableUniversalSceneContractId(
              `${args.provider}:${args.model}:${args.context.learner_message}:${director.contract_id}:${JSON.stringify(universalRecord).slice(0, 600)}`,
            ),
      title: universalRecord.title ?? director.title,
      learner_signal: args.context.learner_message,
      diagnosis_guess:
        universalRecord.diagnosis_guess ??
        director.learning_context.root_problem ??
        "The learner needs the hidden relationship turned into visible motion.",
      learning_goal:
        universalRecord.learning_goal ??
        director.creative_brief.why_this_should_unstick_the_learner ??
        "Show the missing link with an animated scene and a checkpoint.",
      director_plan: universalRecord.director_plan ?? {
        visual_intent: director.creative_brief.visual_metaphor,
        misconception_target: director.learning_context.misconception_target ?? "The current mental model is missing a relationship.",
        key_relationships: director.relationships.map((relationship) => relationship.explanation).slice(0, 5),
        design_principles: [
          director.creative_brief.aha_moment,
          "Use motion to reveal the hidden relationship.",
          "Keep personalization clarifying rather than decorative.",
        ],
      },
      checkpoint: universalRecord.checkpoint ?? director.checkpoint,
    },
    args.context.learner_message,
  );
}

function normalizeProviderDraft(args: {
  draft: unknown;
  model: string;
  provider: UniversalSceneProvider;
  context: VideoDirectorRequestContext;
}): { contract: UniversalSceneContract; director_contract: MyWayVideoDirectorContract } {
  const draftRecord = asRecord(args.draft);
  const directorDraft =
    draftRecord.director_contract ??
    draftRecord.directorContract ??
    draftRecord.video_director_contract ??
    draftRecord.videoDirectorContract ??
    draftRecord;

  const directorRecord = asRecord(directorDraft);
  const existingContractId = directorRecord.contract_id;

  const director_contract = normalizeVideoDirectorContract(
    {
      ...directorRecord,
      schema_version: "myway_video_director_contract_v1",
      contract_id:
        typeof existingContractId === "string" && existingContractId.trim()
          ? existingContractId
          : stableVideoDirectorContractId(`${args.provider}:${args.model}:${args.context.learner_message}:${JSON.stringify(draftRecord).slice(0, 700)}`),
    },
    args.context,
  );

  const contract = buildUniversalSceneFromDirector({
    draftRecord,
    directorContract: director_contract,
    context: args.context,
    provider: args.provider,
    model: args.model,
  });

  return { contract, director_contract };
}

async function callOllama(args: { context: VideoDirectorRequestContext; model: string }): Promise<ProviderGenerationResult> {
  const startedAt = Date.now();

  const response = await fetchWithTimeout(
    `${OLLAMA_URL}/api/chat`,
    {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({
        model: args.model,
        stream: false,
        keep_alive: "10m",
        format: "json",
        options: {
          temperature: 0.12,
          top_p: 0.85,
          num_ctx: 4096,
          num_predict: 2600,
        },
        messages: [
          { role: "system", content: buildSystemPrompt() },
          { role: "user", content: buildUserPrompt(args.context) },
        ],
      }),
    },
    REQUEST_TIMEOUT_MS,
  );

  const elapsed_ms = Date.now() - startedAt;

  if (!response.ok) {
    const detail = await response.text().catch(() => "");
    throw new Error(`Ollama request failed (${response.status}) after ${elapsed_ms}ms. ${detail.slice(0, 640)}`);
  }

  const data = (await response.json()) as OllamaChatResponse;
  const content = contentToString(data?.message?.content);

  if (!content.trim()) {
    throw new Error(`Ollama response did not include usable message.content after ${elapsed_ms}ms.`);
  }

  const draft = safeParseJson(content);
  const normalized = normalizeProviderDraft({
    draft,
    provider: "ollama",
    model: args.model,
    context: args.context,
  });

  return {
    ...normalized,
    meta: {
      elapsed_ms,
      done_reason: data.done_reason ?? null,
      prompt_eval_count: data.prompt_eval_count ?? null,
      eval_count: data.eval_count ?? null,
      raw_content_preview: content.slice(0, 1100),
    },
  };
}

async function callNvidia(args: { context: VideoDirectorRequestContext; model: string }): Promise<ProviderGenerationResult> {
  const apiKey = process.env.NVIDIA_API_KEY;

  if (!apiKey?.trim()) {
    throw new Error("Missing NVIDIA_API_KEY. Add it to .env.local and restart pnpm dev.");
  }

  const startedAt = Date.now();

  const response = await fetchWithTimeout(
    `${NVIDIA_BASE_URL}/chat/completions`,
    {
      method: "POST",
      headers: {
        authorization: `Bearer ${apiKey}`,
        "content-type": "application/json",
        accept: "application/json",
      },
      body: JSON.stringify({
        model: args.model,
        temperature: 0,
        top_p: 0.82,
        max_tokens: 3600,
        response_format: { type: "json_object" },
        chat_template_kwargs: {
          enable_thinking: false,
        },
        messages: [
          { role: "system", content: buildSystemPrompt() },
          { role: "user", content: buildUserPrompt(args.context) },
        ],
      }),
    },
    REQUEST_TIMEOUT_MS,
  );

  const elapsed_ms = Date.now() - startedAt;

  if (!response.ok) {
    const detail = await response.text().catch(() => "");
    throw new Error(`NVIDIA request failed (${response.status}) after ${elapsed_ms}ms. ${detail.slice(0, 900)}`);
  }

  const data = (await response.json()) as NvidiaChatResponse;
  const content = contentToString(data?.choices?.[0]?.message?.content);

  if (!content.trim()) {
    throw new Error(`NVIDIA response did not include usable choices[0].message.content after ${elapsed_ms}ms.`);
  }

  const draft = safeParseJson(content);
  const normalized = normalizeProviderDraft({
    draft,
    provider: "nvidia",
    model: args.model,
    context: args.context,
  });

  return {
    ...normalized,
    meta: {
      elapsed_ms,
      finish_reason: data.choices?.[0]?.finish_reason ?? null,
      prompt_tokens: data.usage?.prompt_tokens ?? null,
      completion_tokens: data.usage?.completion_tokens ?? null,
      total_tokens: data.usage?.total_tokens ?? null,
      raw_content_preview: content.slice(0, 1400),
    },
  };
}

function normalizeProvider(value: unknown, model: string): UniversalSceneProvider {
  if (value === "nvidia" || model.startsWith("nvidia/")) return "nvidia";
  return "ollama";
}

function formatProviderError(error: unknown, provider: UniversalSceneProvider) {
  if (error instanceof Error) {
    if (error.name === "AbortError" || /aborted/i.test(error.message)) {
      return `${provider} video-director generation timed out or was aborted after ${Math.round(REQUEST_TIMEOUT_MS / 1000)}s.`;
    }
    return error.message;
  }

  return `Unknown ${provider} error`;
}

export async function GET() {
  const startedAt = Date.now();

  let ollamaStatus: unknown = null;

  try {
    const response = await fetchWithTimeout(`${OLLAMA_URL}/api/tags`, { method: "GET" }, PING_TIMEOUT_MS);
    const text = await response.text();
    ollamaStatus = {
      ok: response.ok,
      status: response.status,
      preview: text.slice(0, 800),
    };
  } catch (error) {
    ollamaStatus = {
      ok: false,
      error: error instanceof Error ? `${error.name}: ${error.message}` : "Unknown Ollama ping error",
    };
  }

  return NextResponse.json({
    ok: true,
    route: "universal-scene-contract",
    mode: "video-director-plus-universal-scene",
    elapsed_ms: Date.now() - startedAt,
    default_context: buildDefaultVideoDirectorRequestContext("I am stuck on this idea."),
    providers: {
      ollama: {
        base_url: OLLAMA_URL,
        default_model: DEFAULT_OLLAMA_MODEL,
        status: ollamaStatus,
      },
      nvidia: {
        base_url: NVIDIA_BASE_URL,
        default_model: DEFAULT_NVIDIA_MODEL,
        has_api_key: Boolean(process.env.NVIDIA_API_KEY?.trim()),
      },
    },
  });
}

export async function POST(request: Request) {
  const rawBody = (await request.json().catch(() => ({}))) as UniversalSceneRequest;
  const learnerMessage =
    rawBody.learner_message?.trim() ||
    rawBody.learner_signal?.trim() ||
    "I do not understand why a baseball runner has to tag up.";

  const context = normalizeVideoDirectorRequestContext({
    ...rawBody,
    learner_message: learnerMessage,
  });

  const requestedModel = rawBody.model?.trim() || "";
  const provider = normalizeProvider(rawBody.provider, requestedModel);
  const model = requestedModel || (provider === "nvidia" ? DEFAULT_NVIDIA_MODEL : DEFAULT_OLLAMA_MODEL);

  try {
    const result =
      provider === "nvidia"
        ? await callNvidia({ context, model })
        : await callOllama({ context, model });

    return NextResponse.json({
      ok: true,
      source: provider,
      provider,
      model,
      contract: result.contract,
      director_contract: result.director_contract,
      context,
      warnings: [
        `${provider} generated a video director contract in ${Math.round(
          Number(result.meta.elapsed_ms ?? 0) / 1000,
        )}s; MyWay normalized it for trusted 3D/hybrid rendering plus 2D fallback.`,
      ],
      debug: result.meta,
    });
  } catch (error) {
    const director_contract = buildFallbackVideoDirectorContract(context);
    const contract = buildFallbackUniversalSceneContract(context.learner_message);
    const message = formatProviderError(error, provider);

    console.error("[myway-video-director-contract]", provider, message);

    return NextResponse.json(
      {
        ok: false,
        source: "fallback",
        provider,
        model,
        error: message,
        contract,
        director_contract,
        context,
        warnings: [
          `${provider} did not return a usable video director contract, so the sandbox returned MyWay's deterministic fallback contracts.`,
          "This route now expects high-level director JSON plus a universal 2D fallback scene. Check the pnpm dev terminal for [myway-video-director-contract].",
        ],
      },
      { status: 200 },
    );
  }
}
