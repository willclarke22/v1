type Tone = NonNullable<MyWayPrimitiveShape["tone"]>;

const toneStyles: Record<Tone, { fill: string; stroke: string; text: string }> = {
  purple: { fill: "rgba(168,85,247,0.28)", stroke: "rgba(221,214,254,0.66)", text: "#faf5ff" },
  blue: { fill: "rgba(14,165,233,0.22)", stroke: "rgba(125,211,252,0.62)", text: "#f0f9ff" },
  green: { fill: "rgba(34,197,94,0.18)", stroke: "rgba(187,247,208,0.48)", text: "#f0fdf4" },

function clamp(value: number, min: number, max: number) {
  return Math.min(max, Math.max(min, value));
}

function easeOutCubic(value: number) {
  const t = clamp(value, 0, 1);
  return 1 - Math.pow(1 - t, 3);
}

function easeInOutSine(value: number) {
  const t = clamp(value, 0, 1);
  return -(Math.cos(Math.PI * t) - 1) / 2;
}

function projectSaddlePoint(x: number, y: number, bend: number) {
  const z = bend * (x * x - y * y);
  const screenX = 640 + (x - y) * 112;
  const screenY = 355 + (x + y) * 36 - z * 72;
  return `${screenX.toFixed(2)},${screenY.toFixed(2)}`;
}

function buildSaddleLines(progress: number) {
  const bend = easeInOutSine(progress) * 0.56;
  const values = [-2, -1.5, -1, -0.5, 0, 0.5, 1, 1.5, 2];
  const samples = 32;
  const lines: string[] = [];

  for (const fixedY of values) {
    const points: string[] = [];
    for (let index = 0; index <= samples; index += 1) {
      const x = -2 + (index / samples) * 4;
      points.push(projectSaddlePoint(x, fixedY, bend));
    }
    lines.push(points.join(" "));
  }

  for (const fixedX of values) {
    const points: string[] = [];
    for (let index = 0; index <= samples; index += 1) {
      const y = -2 + (index / samples) * 4;
      points.push(projectSaddlePoint(fixedX, y, bend));
    }
    lines.push(points.join(" "));
  }

  return lines;
}

function AnimatedText({ children, delayFrames = 0 }: { children: string; delayFrames?: number }) {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const enter = spring({ frame: Math.max(0, frame - delayFrames), fps, config: { damping: 18, stiffness: 90 } });

  return (
    <div
      style={{
        opacity: interpolate(enter, [0, 1], [0, 1]),
        transform: `translateY(${interpolate(enter, [0, 1], [18, 0])}px)`,
      }}
    >
      {children}
    </div>
  );
}

function Shape({ shape, index, progress }: { shape: MyWayPrimitiveShape; index: number; progress: number }) {
  const tone = toneStyles[shape.tone ?? "purple"];
  const width = shape.width ?? 160;
  const height = shape.height ?? 72;
  const appear = clamp((progress - index * 0.08) / 0.45, 0, 1);
  const eased = easeOutCubic(appear);
  const opacity = interpolate(eased, [0, 1], [0, 1]);
  const y = shape.y + interpolate(eased, [0, 1], [24, 0]);

  if (shape.kind === "arrow") {
    const lineWidth = width;
    return (
      <g opacity={opacity}>
        <line x1={shape.x} y1={shape.y} x2={shape.x + lineWidth} y2={shape.y} stroke="rgba(255,255,255,0.58)" strokeWidth="7" strokeLinecap="round" />
        <path d={`M ${shape.x + lineWidth - 20} ${shape.y - 14} L ${shape.x + lineWidth + 3} ${shape.y} L ${shape.x + lineWidth - 20} ${shape.y + 14}`} fill="none" stroke="rgba(255,255,255,0.58)" strokeWidth="7" strokeLinecap="round" strokeLinejoin="round" />
      </g>
    );
  }

  if (shape.kind === "highlight") {
    return (
      <g opacity={opacity}>
        <rect x={shape.x} y={y} width={width} height={height} rx="34" fill="rgba(168,85,247,0.12)" stroke="rgba(221,214,254,0.68)" strokeWidth="4" strokeDasharray="14 12" />
        <text x={shape.x + width / 2} y={y + height / 2 + 9} textAnchor="middle" fill="white" fontSize="32" fontWeight="900">
          {shape.label}
        </text>
      </g>
    );
  }

  if (shape.kind === "bubble") {
    return (
      <g opacity={opacity}>
        <rect x={shape.x} y={y} width={width} height={height} rx="40" fill={tone.fill} stroke={tone.stroke} strokeWidth="4" />
        <text x={shape.x + width / 2} y={y + height / 2 + 10} textAnchor="middle" fill={tone.text} fontSize="32" fontWeight="900">
          {shape.label}
        </text>
      </g>
    );
  }

  return (
    <g opacity={opacity}>
      <rect x={shape.x} y={y} width={width} height={height} rx="26" fill={tone.fill} stroke={tone.stroke} strokeWidth="3" />
      <text x={shape.x + width / 2} y={y + height / 2 + 9} textAnchor="middle" fill={tone.text} fontSize="30" fontWeight="900">
        {shape.label}
      </text>
    </g>
  );
}

function PrimitiveScene({ scene, progress }: { scene: MyWayRemotionScene; progress: number }) {
  return (
    <svg viewBox="0 0 1280 720" style={{ width: "100%", height: "100%" }}>
      <defs>
        <radialGradient id="primitiveGlow" cx="50%" cy="45%" r="62%">
          <stop offset="0%" stopColor="rgba(168,85,247,0.25)" />
          <stop offset="100%" stopColor="rgba(0,0,0,0)" />
        </radialGradient>
      </defs>
      <rect width="1280" height="720" fill="url(#primitiveGlow)" />
      {(scene.shapes ?? []).map((shape, index) => (
        <Shape key={shape.id} shape={shape} index={index} progress={progress} />
      ))}
    </svg>
  );
}

function CardScene({ scene, progress }: { scene: MyWayRemotionScene; progress: number }) {
  const eased = easeOutCubic(progress);
  const second = clamp((progress - 0.28) / 0.55, 0, 1);
  const third = clamp((progress - 0.5) / 0.48, 0, 1);

  const visual = scene.visualKind;
  const labels =
    visual === "claim_evidence"
      ? ["claim", "evidence", "support"]
      : visual === "equation_balance"
        ? ["left side", "=", "right side"]
        : visual === "transfer_flow"
          ? ["giver", "electron", "receiver"]
          : visual === "sequence_steps"
            ? ["first", "then", "result"]
            : ["idea", "link", "try again"];

  return (
    <svg viewBox="0 0 1280 720" style={{ width: "100%", height: "100%" }}>
      <rect width="1280" height="720" fill="rgba(0,0,0,0.02)" />
      <rect x="110" y="235" width="250" height="120" rx="34" fill="rgba(125,211,252,0.18)" stroke="rgba(125,211,252,0.55)" strokeWidth="4" opacity={eased} />
      <text x="235" y="306" textAnchor="middle" fill="white" fontSize="36" fontWeight="900" opacity={eased}>{labels[0]}</text>

      <line x1="390" y1="295" x2={390 + 210 * second} y2="295" stroke="rgba(255,255,255,0.55)" strokeWidth="8" strokeLinecap="round" />
      <path d={`M ${584 + 20 * second} 275 L ${622 + 20 * second} 295 L ${584 + 20 * second} 315`} fill="none" stroke="rgba(255,255,255,0.55)" strokeWidth="8" strokeLinecap="round" strokeLinejoin="round" opacity={second} />

      <rect x="640" y="235" width="250" height="120" rx="34" fill="rgba(221,214,254,0.2)" stroke="rgba(221,214,254,0.62)" strokeWidth="4" opacity={second} />
      <text x="765" y="306" textAnchor="middle" fill="white" fontSize="36" fontWeight="900" opacity={second}>{labels[1]}</text>

      <rect x="945" y={235 - 40 * third} width="250" height="120" rx="34" fill="rgba(34,197,94,0.18)" stroke="rgba(187,247,208,0.5)" strokeWidth="4" opacity={third} />
      <text x="1070" y={306 - 40 * third} textAnchor="middle" fill="white" fontSize="36" fontWeight="900" opacity={third}>{labels[2]}</text>

      {visual === "equation_balance" ? (
        <path d="M 440 500 C 540 550 740 550 840 500" fill="none" stroke="rgba(255,255,255,0.4)" strokeWidth="10" strokeLinecap="round" />
      ) : null}
    </svg>
  );
}

function SceneRenderer({ scene, progress }: { scene: MyWayRemotionScene; progress: number }) {
  if (scene.visualKind === "saddle_surface") return <SaddleScene progress={progress} />;
  if (scene.visualKind === "primitive_scene_graph") return <PrimitiveScene scene={scene} progress={progress} />;
  return <CardScene scene={scene} progress={progress} />;
}
