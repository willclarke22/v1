export function seconds(value: number) {
  return Math.round(value * ENGINE_REEL_FPS);
}

export function stableContractId(input: string) {
  let hash = 0;
  for (let index = 0; index < input.length; index += 1) {
    hash = (hash * 31 + input.charCodeAt(index)) >>> 0;
  }
  return `engine_reel_contract_${hash.toString(16).padStart(8, "0")}`;
}

function asText(value: unknown, fallback: string, maxLength = 240) {
  if (typeof value !== "string") return fallback;
  const trimmed = value.trim().replace(/\s+/g, " ");
  if (!trimmed) return fallback;
  return trimmed.slice(0, maxLength);
}

function asStringArray(value: unknown, fallback: string[], maxItems = 8) {
  if (!Array.isArray(value)) return fallback;
  const cleaned = value
    .map((item) => asText(item, "", 140))
    .filter(Boolean)
    .slice(0, maxItems);
  return cleaned.length ? cleaned : fallback;
}

function asSceneKind(value: unknown, fallback: EngineReelSceneKind): EngineReelSceneKind {
  return typeof value === "string" && ENGINE_REEL_SCENE_KINDS.includes(value as EngineReelSceneKind)
    ? (value as EngineReelSceneKind)
    : fallback;
}

function asDuration(value: unknown, fallback: number) {
  const numberValue = typeof value === "number" ? value : Number(value);
  if (!Number.isFinite(numberValue)) return fallback;
  return Math.max(seconds(2.5), Math.min(seconds(7), Math.round(numberValue)));
}

function normalizeScenes(value: unknown): EngineReelScene[] {
  const fallback = buildDefaultEngineScenes();
  if (!Array.isArray(value)) return fallback;

  const cleaned = value
    .slice(0, 8)
    .map((raw, index): EngineReelScene | null => {
      if (!raw || typeof raw !== "object") return null;
      const record = raw as Record<string, unknown>;
      const defaultKind = ENGINE_REEL_SCENE_KINDS[Math.min(index, ENGINE_REEL_SCENE_KINDS.length - 1)] ?? "overview";
      const kind = asSceneKind(record.kind, defaultKind);
      return {
        id: asText(record.id, `${kind}_${index + 1}`, 40).replace(/[^a-z0-9_-]/gi, "_").toLowerCase(),
        kind,
        title: asText(record.title, kind.replaceAll("_", " "), 52),
        caption: asText(record.caption, "Watch how the timed motion changes the system.", 180),
        durationInFrames: asDuration(record.durationInFrames, seconds(4.5)),
        focus: asText(record.focus, "Focus on the part that changes in this scene.", 120),
      };
    })
    .filter((scene): scene is EngineReelScene => Boolean(scene));

  if (cleaned.length < 3) return fallback;
  if (!cleaned.some((scene) => scene.kind === "checkpoint")) {
    cleaned.push({
      id: "checkpoint",
      kind: "checkpoint",
      title: "Checkpoint",
      caption: "Explain the one motion or timing link that makes the system work.",
      durationInFrames: seconds(4),
      focus: "Ask the learner to name the causal link.",
    });
  }
  return cleaned;
}

function normalizeParts(value: unknown) {
  const fallback = buildDefaultEngineParts();
  if (!Array.isArray(value)) return fallback;
  const cleaned = value
    .slice(0, 10)
    .map((raw, index): EngineReelPart | null => {
      if (!raw || typeof raw !== "object") return null;
      const record = raw as Record<string, unknown>;
      const label = asText(record.label, fallback[index]?.label ?? `part ${index + 1}`, 42);
      return {
        id: asText(record.id, label, 40).replace(/[^a-z0-9_-]/gi, "_").toLowerCase(),
        label,
        role: asText(record.role, fallback[index]?.role ?? "helps the timed system work", 120),
      };
    })
    .filter((part): part is EngineReelPart => Boolean(part));
  return cleaned.length >= 4 ? cleaned : fallback;
}

      return {

  const learnerSignal = asText(record.learner_signal, "I do not get how a gas engine makes the piston move.", 360);
  const fallback = {
    source_style:
      "Clean educational cutaway animation: fixed camera, moving parts, colored flow, labels, captions, and checkpoint.",
    recreated_elements: [
      "fixed cutaway view",
      "moving piston",
      "timed valves",
      "colored flow",
      "spark or trigger moment",
      "caption layer",
    ],
    intentionally_not_copied: [
      "no original frames used",
      "new MyWay visual treatment",
      "contract-directed rendering only",
    ],
  };
  const visualReference = record.visual_reference && typeof record.visual_reference === "object"
    ? (record.visual_reference as Record<string, unknown>)
    : {};
  const checkpoint = record.checkpoint && typeof record.checkpoint === "object"
    ? (record.checkpoint as Record<string, unknown>)
    : {};

  return {
    schema_version: "myway_engine_reel_remotion_contract_v1",
    contract_id: asText(record.contract_id, stableContractId(JSON.stringify(record).slice(0, 500)), 72),
    title: asText(record.title, "A visual explanation built from your message", 70),

      "Show the missing link with a focused animation and one checkpoint.",

    ),
    format: {
      width: ENGINE_REEL_WIDTH,
      height: ENGINE_REEL_HEIGHT,
      fps: ENGINE_REEL_FPS,
      aspect: "16:9_laptop",
    },
    visual_reference: {

    },
    remotion_strategy: buildFixedRemotionStrategy(),
    parts: normalizeParts(record.parts),
    motions: normalizeMotions(record.motions),

    checkpoint: {

    },

export function getEngineReelDurationInFrames(contract: EngineReelContract) {
  return contract.scenes.reduce((sum, scene) => sum + scene.durationInFrames, 0);
}

export function findEngineReelScene(contract: EngineReelContract, frame: number) {
  let cursor = 0;

  for (let index = 0; index < contract.scenes.length; index += 1) {
    const scene = contract.scenes[index]!;
    const startFrame = cursor;
    const endFrame = startFrame + scene.durationInFrames;

    if (frame <= endFrame || index === contract.scenes.length - 1) {
      return {
        scene,
        sceneIndex: index,
        startFrame,
        endFrame,
        progress: Math.max(0, Math.min(1, (frame - startFrame) / Math.max(1, scene.durationInFrames))),
      };
    }

    cursor = endFrame;
  }

  const fallback = contract.scenes[0]!;
  return {
    scene: fallback,
    sceneIndex: 0,
    startFrame: 0,
    endFrame: fallback.durationInFrames,
    progress: 0,
  };
}
