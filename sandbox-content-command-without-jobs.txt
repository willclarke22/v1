cd C:\Users\willc\projects\MyWay\v1

$ProjectRoot = (Resolve-Path -LiteralPath ".").Path
$Out = Join-Path $ProjectRoot "myway-sandbox-active-files-one-notepad.txt"

if (Test-Path -LiteralPath $Out) {
  Remove-Item -LiteralPath $Out -Force
}

$TextExtensions = @(
  ".ts", ".tsx", ".js", ".jsx", ".cjs", ".mjs",
  ".json", ".md", ".txt", ".py", ".css", ".scss",
  ".html", ".yml", ".yaml"
)

function Is-TextFile {
  param(
    [Parameter(Mandatory)]
    [System.IO.FileInfo]$File
  )

  return $TextExtensions -contains $File.Extension.ToLowerInvariant()
}

function Add-Line {
  param(
    [AllowEmptyString()]
    [string]$Text = ""
  )

  Add-Content `
    -LiteralPath $Out `
    -Value $Text `
    -Encoding UTF8
}

function Get-NormalizedRelativePath {
  param(
    [Parameter(Mandatory)]
    [System.IO.FileInfo]$File
  )

  return [System.IO.Path]::GetRelativePath(
    $ProjectRoot,
    $File.FullName
  ).Replace("/", "\")
}

function Is-ActiveSandboxFile {
  param(
    [Parameter(Mandatory)]
    [System.IO.FileInfo]$File
  )

  $RelativePath = Get-NormalizedRelativePath $File

  # Include only the active sandbox source locations.
  $IsActiveSandboxPath =
    $RelativePath -like "sandbox\*" -or
    $RelativePath -like "app\sandbox\*" -or
    $RelativePath -like "app\api\sandbox\*" -or
    $RelativePath -like "lib\sandbox\*"

  if (-not $IsActiveSandboxPath) {
    return $false
  }

  # Exclude large or generated runtime job files.
  if (
    $RelativePath -eq "sandbox\probe-lab\assets\jobs" -or
    $RelativePath -like "sandbox\probe-lab\assets\jobs\*"
  ) {
    return $false
  }

  # Defensive exclusions. These should already be outside the explicit
  # active roots, but keeping them here prevents accidental future inclusion.
  $ExcludedDirectoryNames = @(
    ".git",
    ".next",
    "node_modules",
    "archive",
    ".venv",
    ".myway-patch-backups"
  )

  $PathParts = $RelativePath -split "\\"

  foreach ($ExcludedDirectoryName in $ExcludedDirectoryNames) {
    if (
      $PathParts -contains $ExcludedDirectoryName
    ) {
      return $false
    }
  }

  return Is-TextFile $File
}

$Files = @(
  Get-ChildItem `
    -LiteralPath $ProjectRoot `
    -Recurse `
    -File `
    -ErrorAction SilentlyContinue |
    Where-Object {
      Is-ActiveSandboxFile $_
    } |
    Sort-Object {
      Get-NormalizedRelativePath $_
    }
)

Add-Line "MYWAY ACTIVE SANDBOX FILES - ONE NOTEPAD FILE"
Add-Line ("Generated: " + (Get-Date).ToString("o"))
Add-Line ("Project root: " + $ProjectRoot)
Add-Line ""
Add-Line "Selection rule:"
Add-Line "- Include active text files under sandbox\."
Add-Line "- Include active text files under app\sandbox\."
Add-Line "- Include active text files under app\api\sandbox\."
Add-Line "- Include active text files under lib\sandbox\."
Add-Line "- Exclude .myway-patch-backups and all historical patch copies."
Add-Line "- Exclude .git, .next, node_modules, archive, and .venv."
Add-Line "- Exclude sandbox\probe-lab\assets\jobs and all files below it."
Add-Line ""
Add-Line ("Included files: " + $Files.Count)
Add-Line ""

Add-Line ("=" * 120)
Add-Line "INDEX"
Add-Line ("=" * 120)

$Counter = 1

foreach ($File in $Files) {
  $RelativePath = Get-NormalizedRelativePath $File

  Add-Line (
    "{0:D3}: {1} [{2} bytes]" -f `
      $Counter,
      $RelativePath,
      $File.Length
  )

  $Counter++
}

Add-Line ""
Add-Line ""
Add-Line ("=" * 120)
Add-Line "FULL FILE CONTENTS"
Add-Line ("=" * 120)

$Counter = 1

foreach ($File in $Files) {
  $RelativePath = Get-NormalizedRelativePath $File

  Add-Line ""
  Add-Line ""
  Add-Line ("#" * 120)
  Add-Line ("FILE {0:D3}: {1}" -f $Counter, $RelativePath)
  Add-Line ("SIZE BYTES: " + $File.Length)
  Add-Line ("#" * 120)
  Add-Line ""

  try {
    $Content = Get-Content `
      -LiteralPath $File.FullName `
      -Raw `
      -ErrorAction Stop

    if ($null -ne $Content) {
      Add-Content `
        -LiteralPath $Out `
        -Value $Content `
        -Encoding UTF8
    }
    else {
      Add-Line "[EMPTY FILE]"
    }
  }
  catch {
    Add-Line ("ERROR READING FILE: " + $_.Exception.Message)
  }

  $Counter++
}

Write-Host ""
Write-Host "Done."
Write-Host "Included active files: $($Files.Count)"
Write-Host ""
Write-Host "Included roots:"
Write-Host "  sandbox\"
Write-Host "  app\sandbox\"
Write-Host "  app\api\sandbox\"
Write-Host "  lib\sandbox\"
Write-Host ""
Write-Host "Excluded:"
Write-Host "  .myway-patch-backups\"
Write-Host "  sandbox\probe-lab\assets\jobs\"
Write-Host "  .git\, .next\, node_modules\, archive\, .venv\"
Write-Host ""
Write-Host "Output:"
Write-Host "  $Out"
Write-Host ""

notepad $Out
