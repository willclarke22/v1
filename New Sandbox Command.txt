function Get-ProjectRelativePath {
    param(
        [Parameter(Mandatory)]
        [string]$FullPath
    )

    $Normalized =
        [System.IO.Path]::GetFullPath(
            $FullPath
        )

    return [System.IO.Path]::GetRelativePath(
        $ProjectRoot,
        $Normalized
    ).Replace("/", "\")
}

function Test-IsInsideProject {
    param(
        [Parameter(Mandatory)]
        [string]$Path
    )

    try {
        $Full =
            [System.IO.Path]::GetFullPath(
                $Path
            )
    }
    catch {
        return $false
    }

    if (
        $Full.Equals(
            $ProjectRoot,
            [System.StringComparison]::OrdinalIgnoreCase
        )
    ) {
        return $true
    }

    return $Full.StartsWith(
        $ProjectPrefix,
        [System.StringComparison]::OrdinalIgnoreCase
    )
}

function Format-Bytes {
    param(
        [long]$Bytes
    )

    if ($Bytes -ge 1GB) {
        return "{0:N2} GB" -f (
            $Bytes / 1GB
        )
    }

    if ($Bytes -ge 1MB) {
        return "{0:N2} MB" -f (
            $Bytes / 1MB
        )
    }

    if ($Bytes -ge 1KB) {
        return "{0:N2} KB" -f (
            $Bytes / 1KB
        )
    }

    return "$Bytes bytes"
}

function Test-IsExcludedProjectPath {
    param(
        [Parameter(Mandatory)]
        [string]$FullPath
    )

    if (-not (Test-IsInsideProject $FullPath)) {
        return $true
    }

    $Relative =
        Get-ProjectRelativePath `
            $FullPath

    $Parts =
        $Relative -split "\\"

    foreach ($Part in $Parts) {
        if (
            $ExcludedDirectoryNames.Contains(
                $Part
            )
        ) {
            return $true
        }
    }

    return $false
}

function Test-IsRuntimeContentPath {
    param(
        [Parameter(Mandatory)]
        [string]$FullPath
    )

    if (-not (Test-IsInsideProject $FullPath)) {
        return $false
    }

    $Relative =
        Get-ProjectRelativePath `
            $FullPath

    foreach ($Prefix in $RuntimeContentExclusions) {

        if (
            $Relative.Equals(
                $Prefix,
                [System.StringComparison]::OrdinalIgnoreCase
            )
        ) {
            return $true
        }

        if (
            $Relative.StartsWith(
                $Prefix + "\",
                [System.StringComparison]::OrdinalIgnoreCase
            )
        ) {
            return $true
        }
    }

    return $false
}

function Test-IsSensitiveFile {
    param(
        [Parameter(Mandatory)]
        [System.IO.FileInfo]$File
    )

    if (
        $File.Name -match '^\.env($|\.)'
    ) {
        return $true
    }

    if (
        $File.Name -match
        '(?i)(secret|credential|private[-_]?key|service[-_]?account)'
    ) {
        return $true
    }

    return $false
}

function Test-IsTextFile {
    param(
        [Parameter(Mandatory)]
        [System.IO.FileInfo]$File
    )

    if (
        Test-IsSensitiveFile $File
    ) {
        return $false
    }

    if (
        $File.Name.EndsWith(
            ".d.ts",
            [System.StringComparison]::OrdinalIgnoreCase
        )
    ) {
        return $true
    }

    return $TextExtensions.Contains(
        $File.Extension
    )
}

# ============================================================================
# DISCOVER EVERY ACTIVE SANDBOX / PROBE-LAB AREA
#
# We discover both names, then collapse nested discoveries so, for example,
# app\sandbox owns app\sandbox\probe-lab rather than scanning both separately.
# ============================================================================

$DiscoveredRoots =
    [System.Collections.Generic.HashSet[string]]::new(
        [System.StringComparer]::OrdinalIgnoreCase
    )

$DiscoveryQueue =
    [System.Collections.Generic.Queue[System.IO.DirectoryInfo]]::new()

$DiscoveryQueue.Enqueue(
    [System.IO.DirectoryInfo]::new(
        $ProjectRoot
    )
)

while (
    $DiscoveryQueue.Count -gt 0
) {
    $Directory =
        $DiscoveryQueue.Dequeue()

    try {
        foreach (
            $Child in
            $Directory.EnumerateDirectories()
        ) {
            if (
                $ExcludedDirectoryNames.Contains(
                    $Child.Name
                )
            ) {
                continue
            }

            $IsSandbox =
                $Child.Name.Equals(
                    "sandbox",
                    [System.StringComparison]::OrdinalIgnoreCase
                )

            $IsProbeLab =
                $Child.Name.Equals(
                    "probe-lab",
                    [System.StringComparison]::OrdinalIgnoreCase
                )

            if (
                $IsSandbox -or
                $IsProbeLab
            ) {
                [void]$DiscoveredRoots.Add(
                    $Child.FullName
                )
            }

            $DiscoveryQueue.Enqueue(
                $Child
            )
        }
    }
    catch {
        continue
    }
}

$CandidateRoots = @(
    $DiscoveredRoots |
    Sort-Object {
        $_.Length
    }
)

$ActiveRootList =
    [System.Collections.Generic.List[string]]::new()

foreach ($Candidate in $CandidateRoots) {

    $NestedUnderExisting =
        $false

    foreach ($Existing in $ActiveRootList) {

        $ExistingPrefix =
            $Existing.TrimEnd("\") + "\"

        if (
            $Candidate.StartsWith(
                $ExistingPrefix,
                [System.StringComparison]::OrdinalIgnoreCase
            )
        ) {
            $NestedUnderExisting =
                $true

            break
        }
    }

    if (-not $NestedUnderExisting) {
        $ActiveRootList.Add(
            $Candidate
        )
    }
}

$ActiveRoots = @(
    $ActiveRootList |
    Sort-Object
)

function Add-SelectedFile {
    param(
        [Parameter(Mandatory)]
        [System.IO.FileInfo]$File,

        [Parameter(Mandatory)]
        [string]$Reason
    )

    if (
        Test-IsExcludedProjectPath `
            $File.FullName
    ) {
        return
    }

    if (
        Test-IsRuntimeContentPath `
            $File.FullName
    ) {
        return
    }

    if (
        -not (
            Test-IsTextFile `
                $File
        )
    ) {
        return
    }

    if (
        $SelectedPaths.Add(
            $File.FullName
        )
    ) {
        $SelectionReason[
            $File.FullName
        ] = $Reason

        if (
            $CodeExtensions.Contains(
                $File.Extension
            )
        ) {
            $ImportQueue.Enqueue(
                $File.FullName
            )
        }
    }
}

# ============================================================================

function Resolve-LocalImport {
    param(
        [Parameter(Mandatory)]
        [string]$ImporterPath,

        [Parameter(Mandatory)]
        [string]$Specifier
    )

    $BasePath = $null

    if (
        $Specifier.StartsWith("@/")
    ) {
        $BasePath = Join-Path `
            $ProjectRoot `
            $Specifier.Substring(2)
    }
    elseif (
        $Specifier.StartsWith(".")
    ) {
        $BasePath = Join-Path `
            (
                Split-Path `
                    -Parent `
                    $ImporterPath
            ) `
            $Specifier
    }
    else {
        return $null
    }

    try {
        $BasePath =
            [System.IO.Path]::GetFullPath(
                $BasePath
            )
    }
    catch {
        return $null
    }

    if (
        Test-IsExcludedProjectPath `
            $BasePath
    ) {
        return $null
    }

    if (
        Test-IsRuntimeContentPath `
            $BasePath
    ) {
        return $null
    }

    $Candidates =
        [System.Collections.Generic.List[string]]::new()

    $Candidates.Add(
        $BasePath
    )

    foreach ($Extension in @(
        ".ts",
        ".tsx",
        ".js",
        ".jsx",
        ".mjs",
        ".cjs",
        ".json",
        ".css",
        ".scss",
        ".py"
    )) {
        $Candidates.Add(
            $BasePath + $Extension
        )

        $Candidates.Add(
            (
                Join-Path `
                    $BasePath `
                    ("index" + $Extension)
            )
        )
    }

    foreach ($Candidate in $Candidates) {

        if (
            -not (
                Test-Path `
                    -LiteralPath $Candidate `
                    -PathType Leaf
            )
        ) {
            continue
        }

        $File =
            Get-Item `
                -LiteralPath $Candidate

        if (
            Test-IsExcludedProjectPath `
                $File.FullName
        ) {
            continue
        }

        if (
            Test-IsRuntimeContentPath `
                $File.FullName
        ) {
            continue
        }

        if (
            -not (
                Test-IsTextFile `
                    $File
            )
        ) {
            continue
        }

        return $File.FullName
    }

    return $null
}

$ImportPatterns = @(
    '(?m)\bfrom\s+["'']([^"'']+)["'']',
    '(?m)\bimport\s*\(\s*["'']([^"'']+)["'']\s*\)',
    '(?m)\brequire\s*\(\s*["'']([^"'']+)["'']\s*\)',
    '(?m)^\s*import\s*["'']([^"'']+)["'']\s*;?'
)

$UnresolvedImports =
    [System.Collections.Generic.HashSet[string]]::new(
        [System.StringComparer]::OrdinalIgnoreCase
    )

while (
    $ImportQueue.Count -gt 0
) {
    $ImporterPath =
        $ImportQueue.Dequeue()

    try {
        $Content =
            [System.IO.File]::ReadAllText(
                $ImporterPath
            )
    }
    catch {
        continue
    }

    foreach ($Pattern in $ImportPatterns) {

        $Matches =
            [System.Text.RegularExpressions.Regex]::Matches(
                $Content,
                $Pattern
            )

        foreach ($Match in $Matches) {

            $Specifier =
                $Match.Groups[1].Value.Trim()

            if (
                -not $Specifier.StartsWith(".") -and
                -not $Specifier.StartsWith("@/")
            ) {
                continue
            }

            $Resolved =
                Resolve-LocalImport `
                    -ImporterPath $ImporterPath `
                    -Specifier $Specifier

            if ($Resolved) {
                Add-SelectedFile `
                    -File (
                        Get-Item `
                            -LiteralPath $Resolved
                    ) `
                    -Reason "local-import"
            }
            else {
                $Description =
                    (
                        Get-ProjectRelativePath `
                            $ImporterPath
                    ) +
                    " :: " +
                    $Specifier

                [void]$UnresolvedImports.Add(
                    $Description
                )
            }
        }
    }
}

$RuntimeFootprint =
    [System.Collections.Generic.List[object]]::new()

foreach (
    $RelativePath in
    $RuntimeContentExclusions
) {
    $FullPath = Join-Path `
        $ProjectRoot `
        $RelativePath

    if (
        -not (
            Test-Path `
                -LiteralPath $FullPath
        )
    ) {
        $RuntimeFootprint.Add(
            [pscustomobject]@{
                Path = $RelativePath
                Present = $false
                FileCount = 0
                Bytes = 0L
            }
        )

        continue
    }

    $Files = @(
        Get-ChildItem `
            -LiteralPath $FullPath `
            -Recurse `
            -File `
            -Force `
            -ErrorAction SilentlyContinue
    )

    $Bytes = 0L

    foreach ($File in $Files) {
        $Bytes +=
            [long]$File.Length
    }

    $RuntimeFootprint.Add(
        [pscustomobject]@{
            Path = $RelativePath
            Present = $true
            FileCount = $Files.Count
            Bytes = $Bytes
        }
    )
}

    # RUNTIME FOOTPRINT
    # ========================================================================

    Add-Line ("=" * 120)
    Add-Line "LOCAL RUNTIME / GENERATED FOOTPRINT"
    Add-Line ("=" * 120)

    foreach ($Row in $RuntimeFootprint) {

        if (-not $Row.Present) {
            Add-Line (
                "{0,-75} | NOT PRESENT" -f
                $Row.Path
            )

            continue
        }

        Add-Line (
            "{0,-75} | files={1,5} | size={2}" -f
            $Row.Path,
            $Row.FileCount,
            (
                Format-Bytes `
                    $Row.Bytes
            )
        )
    }

    Add-Line ""

    # ========================================================================
    # SOURCE INDEX
    # ========================================================================

    Add-Line ("=" * 120)
    Add-Line "SELECTED SOURCE FILE INDEX"
    Add-Line ("=" * 120)

    $Counter = 1

    foreach ($File in $SelectedFiles) {

        $Relative =
            Get-ProjectRelativePath `
                $File.FullName

        $GitLabel =
            "untracked"

        if (
            $GitTracked.Contains(
                $Relative
            )
        ) {
            $GitLabel =
                "tracked"
        }

        $Reason =
            [string]$SelectionReason[
                $File.FullName
            ]

        Add-Line (
            "{0:D4}: {1} [{2}; {3}; {4}]" -f
            $Counter,
            $Relative,
            (
                Format-Bytes `
                    $File.Length
            ),
            $GitLabel,
            $Reason
        )

        $Counter++
    }

    Add-Line ""

    # ========================================================================
    # UNRESOLVED IMPORTS
    # ========================================================================

    Add-Line ("=" * 120)
    Add-Line "UNRESOLVED LOCAL IMPORTS"
    Add-Line ("=" * 120)

    if (
        $UnresolvedImports.Count -eq 0
    ) {
        Add-Line "None."
    }
    else {
        foreach (
            $Import in
            @(
                $UnresolvedImports |
                Sort-Object
            )
        ) {
            Add-Line $Import
        }
    }

    Add-Line ""

    # ========================================================================
    # FULL SOURCE CONTENTS
    # ========================================================================

    Add-Line ("=" * 120)
    Add-Line "FULL SELECTED SOURCE CONTENTS"
    Add-Line ("=" * 120)

    $Counter = 1

    foreach ($File in $SelectedFiles) {

        $Relative =
            Get-ProjectRelativePath `
                $File.FullName

        $GitLabel =
            "untracked"

        if (
            $GitTracked.Contains(
                $Relative
            )
        ) {
            $GitLabel =
                "tracked"
        }

        $Reason =
            [string]$SelectionReason[
                $File.FullName
            ]

        Add-Line ""
        Add-Line ""
        Add-Line ("#" * 120)

        Add-Line (
            "FILE {0:D4}: {1}" -f
            $Counter,
            $Relative
        )

        Add-Line (
            "SIZE: " +
            (
                Format-Bytes `
                    $File.Length
            )
        )

        Add-Line (
            "GIT: " +
            $GitLabel
        )

        Add-Line (
            "SELECTION: " +
            $Reason
        )

        Add-Line ("#" * 120)
        Add-Line ""

        if (
            $File.Length -gt
            $MaxEmbeddedFileBytes
        ) {
            Add-Line (
                "[CONTENT OMITTED: file exceeds " +
                (
                    Format-Bytes `
                        $MaxEmbeddedFileBytes
                ) +
                "]"
            )
        }
        else {
            try {
                $Text =
                    [System.IO.File]::ReadAllText(
                        $File.FullName
                    )

                Add-RawText `
                    $Text
            }
            catch {
                Add-Line (
                    "ERROR READING FILE: " +
                    $_.Exception.Message
                )
            }
        }

        $Counter++
    }

    Add-Line ""

    # ========================================================================
    # END SUMMARY
    # ========================================================================

    Add-Line ("=" * 120)
    Add-Line "END SUMMARY"
    Add-Line ("=" * 120)

    Add-Line (
        "Top-level sandbox/probe-lab roots: " +
        $ActiveRoots.Count
    )

    Add-Line (
        "Filesystem inventory files: " +
        $InventoryFiles.Count
    )

    Add-Line (
        "Selected source/import/config files: " +
        $SelectedFiles.Count
    )

    Add-Line (
        "Unresolved local imports: " +
        $UnresolvedImports.Count
    )

    $RuntimeFileCount = 0
    $RuntimeBytes = 0L

    foreach ($Row in $RuntimeFootprint) {
        $RuntimeFileCount +=
            [int]$Row.FileCount

        $RuntimeBytes +=
            [long]$Row.Bytes
    }

    Add-Line (
        "Known runtime/generated files: " +
        $RuntimeFileCount
    )

    Add-Line (
        "Known runtime/generated size: " +
        (
            Format-Bytes `
                $RuntimeBytes
        )
    )

    Add-Line ""
}
finally {
    $script:Writer.Dispose()
}

Write-Host "=============================================================="
Write-Host ""
Write-Host "Output:"
Write-Host "  $Out"

    notepad $Out
}
